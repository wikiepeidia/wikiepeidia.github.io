```bash

hugo version
# Verify output contains "extended"
```

**Step 2: Initialize Hugo in the existing repo**
The repo has existing files (README.md, LICENSE, MAINREADME.md) so use --force:
```bash
hugo new site . --force
```
This creates: archetypes/, assets/, content/, layouts/, static/, themes/ directories and hugo.toml placeholder. Existing files are NOT touched.

**Step 3: Add hugo-theme-terminal as git submodule**
```bash
git submodule add https://github.com/panr/hugo-theme-terminal themes/terminal
git submodule update --init --recursive
```
This creates `.gitmodules` with the submodule reference.

**Step 4: Copy theme archetype to project**
```bash
cp themes/terminal/archetypes/posts.md archetypes/posts.md
```
This gives the project its own posts archetype that overrides the theme's. Do not edit yet.

**Step 5: Commit scaffold**
```bash
git add archetypes/ assets/ content/ layouts/ static/ themes/ .gitmodules
git commit -m "feat(01-foundation): initialize Hugo site with terminal theme submodule"
```

**Step 1: Replace hugo.toml with complete configuration**

Overwrite the entire hugo.toml with the following (verbatim):

```toml
baseURL = "https://wikiepeidia.github.io/"
languageCode = "en-us"
defaultContentLanguage = "en"
theme = "terminal"
paginate = 5
title = "wikiepeidia"

[params]
  themeColor = "orange"
  fullWidthTheme = false
  centerTheme = true
  contentTypeName = "posts"
  autoCover = true
  readingTime = true
  showFullContent = false

[params.twitter]
  creator = "@wikiepeidia"

[menu]
  [[menu.main]]
    identifier = "blog"
    name = "Blog"
    url = "/posts"
    weight = 1
  [[menu.main]]
    identifier = "about"
    name = "About"
    url = "/about"
    weight = 2
  [[menu.main]]
    identifier = "github-stats"
    name = "GitHub Stats"
    url = "/github-stats"
    weight = 3

[markup]
  [markup.goldmark]
    [markup.goldmark.renderer]
      unsafe = true
  [markup.highlight]
    noClasses = false
    codeFences = true
    guessSyntax = true
    lineNos = false
    style = "monokai"

[outputs]
  home = ["HTML", "RSS"]
  section = ["HTML", "RSS"]
  term = ["HTML", "RSS"]

[outputFormats]
  [outputFormats.RSS]
    mediatype = "application/rss+xml"
    baseName = "index"
```

**Step 2: Write .gitignore**

Create/overwrite `.gitignore`:
```
# Hugo build output — generated by CI, never commit
/public/

# Hugo resource cache
/resources/_gen/

# OS files
.DS_Store
Thumbs.db
```

**Step 3: Verify local build**

Run Hugo build (not server, to avoid needing a browser):
```bash
hugo --minify 2>&1
# Must exit 0 with no errors
# Expected output: "Total in Xs" line
```

If hugo server is preferred for UI verification:
```bash
hugo server --disableFastRender 2>&1 &
sleep 3
curl -s http://localhost:1313 | grep -i "terminal" | head -5
kill %1
```

**Step 4: Ensure public/ is NOT tracked**
```bash
git rm -r --cached public/ 2>/dev/null || true
```

**Step 5: Commit**
```bash

git commit -m "feat(01-foundation): add hugo.toml config and .gitignore"
```
